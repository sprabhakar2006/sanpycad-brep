# SanPyCAD Brep is B-rep only: every script builds real build123d/OCCT
# solids (see backend/brep.py), so build123d is a hard dependency here,
# unlike the mesh-based SanPyCAD apps. numpy is still needed for the
# viewer JSON and STL/OBJ/SVG/DXF export triangle math.
numpy
build123d

# build123d's own topology/shape_core.py does a plain top-level
# `from IPython.lib.pretty import ...` (used for its pretty-printed
# repr, nothing Jupyter-specific), so importing build123d at all fails
# without IPython installed -- it is not actually optional, despite
# only being needed for notebook-style pretty printing elsewhere.
ipython

# Needed only if a script does `import ocad` -- the opt-in mesh/
# point-list escape hatch (see backend/python_eval.py's module
# docstring). A copy of ocad.py is bundled in backend/, but it
# still needs these to actually import; several examples use it
# (02_ocad_bridge_fillet.py, 05_woven_lamp_shade.py,
# 06_mobile_stand.py, 07_flanged_bracket_assembly.py). Skip these if
# you never plan to write scripts that reach for that toolkit.
# scipy is pinned below 1.14 for consistency with build_bundle.py's
# install list, though the actual packaged-app crash this used to be
# blamed on ("NameError: name 'obj' is not defined" deep in
# scipy/stats/_distn_infrastructure.py) turned out to be unrelated to
# scipy's version: it's a CPython 3.12.0 bug in code.replace() (fixed
# in 3.12.1) triggered by PyInstaller's own path-anonymization step on
# scipy's bytecode -- see packaging/_run_pyinstaller_patched.py for
# the real fix and full writeup. Running from source (this
# requirements.txt's normal use) was never affected either way.
scipy<1.14
sympy
scikit-image

# Optional, for a real desktop app window instead of opening a browser tab:
pywebview
